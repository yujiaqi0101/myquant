### 4.1 相似度计算方法

#### 4.1.1 基于价格形态的相似度（传统方法）

**1. 标准化价格序列**
```python
def normalize_price(prices):
    """Z-score标准化"""
    return (prices - np.mean(prices)) / np.std(prices)

# 或 Min-Max标准化
def minmax_normalize(prices):
    return (prices - np.min(prices)) / (np.max(prices) - np.min(prices))
```

**2. 动态时间规整（DTW）**
```python
from scipy.spatial.distance import euclidean
from fastdtw import fastdtw

def calculate_dtw_similarity(series1, series2):
    """
    计算两个时间序列的DTW距离
    优点：可以处理不同长度和速度变化的序列
    """
    distance, path = fastdtw(series1, series2, dist=euclidean)
    similarity = 1 / (1 + distance)
    return similarity
```

**3. 余弦相似度**
```python
def cosine_similarity(series1, series2):
    """计算余弦相似度"""
    return np.dot(series1, series2) / (np.linalg.norm(series1) * np.linalg.norm(series2))
```

### 4.2 深度学习相似度算法（新增）

#### 4.2.1 LSTM自编码器特征提取

```python
import torch
import torch.nn as nn

class LSTMAutoencoder(nn.Module):
    """
    LSTM自编码器 - 用于提取股价走势的深层特征
    思路：将价格序列压缩为低维特征向量，用于相似度计算
    """
    
    def __init__(self, input_dim=1, hidden_dim=64, latent_dim=16, seq_len=20):
        super().__init__()
        
        # 编码器
        self.encoder = nn.LSTM(
            input_size=input_dim,
            hidden_size=hidden_dim,
            num_layers=2,
            batch_first=True,
            dropout=0.2
        )
        
        # 潜在空间
        self.fc_encode = nn.Linear(hidden_dim, latent_dim)
        
        # 解码器
        self.fc_decode = nn.Linear(latent_dim, hidden_dim)
        self.decoder = nn.LSTM(
            input_size=hidden_dim,
            hidden_size=hidden_dim,
            num_layers=2,
            batch_first=True,
            dropout=0.2
        )
        self.output_layer = nn.Linear(hidden_dim, input_dim)
        
    def encode(self, x):
        """编码：序列 -> 潜在向量"""
        _, (hidden, _) = self.encoder(x)
        # 取最后一层的隐状态
        latent = self.fc_encode(hidden[-1])
        return latent
    
    def decode(self, latent):
        """解码：潜在向量 -> 序列"""
        hidden = self.fc_decode(latent)
        hidden = hidden.unsqueeze(1).repeat(1, seq_len, 1)
        output, _ = self.decoder(hidden)
        output = self.output_layer(output)
        return output
    
    def forward(self, x):
        latent = self.encode(x)
        reconstructed = self.decode(latent)
        return reconstructed, latent


def train_autoencoder(price_sequences, epochs=100):
    """
    训练自编码器
    price_sequences: shape (n_samples, seq_len, 1)
    """
    model = LSTMAutoencoder()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
    criterion = nn.MSELoss()
    
    model.train()
    for epoch in range(epochs):
        for batch in price_sequences:
            optimizer.zero_grad()
            reconstructed, latent = model(batch)
            loss = criterion(reconstructed, batch)
            loss.backward()
            optimizer.step()
    
    return model


def calculate_dl_similarity(model, series1, series2):
    """
    使用深度学习模型计算相似度
    思路：通过自编码器提取潜在特征，计算特征向量的余弦相似度
    """
    model.eval()
    with torch.no_grad():
        latent1 = model.encode(series1)
        latent2 = model.encode(series2)
    
    # 计算潜在向量的余弦相似度
    similarity = torch.nn.functional.cosine_similarity(latent1, latent2)
    return similarity.item()
```

#### 4.2.2 时序卷积网络（TCN）相似度

```python
class TCNEncoder(nn.Module):
    """
    时序卷积网络编码器
    优点：并行计算效率高，感受野可控
    """
    
    def __init__(self, input_dim=1, num_channels=[32, 64, 128], kernel_size=3):
        super().__init__()
        
        layers = []
        for i in range(len(num_channels)):
            in_ch = input_dim if i == 0 else num_channels[i-1]
            out_ch = num_channels[i]
            layers.append(nn.Conv1d(in_ch, out_ch, kernel_size, padding=kernel_size//2))
            layers.append(nn.ReLU())
            layers.append(nn.BatchNorm1d(out_ch))
        
        self.network = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool1d(1)
    
    def forward(self, x):
        # x: (batch, seq_len, features) -> (batch, features, seq_len)
        x = x.transpose(1, 2)
        x = self.network(x)
        x = self.pool(x).squeeze(-1)  # (batch, features)
        return x


def calculate_tcn_similarity(model, series1, series2):
    """使用TCN编码器计算相似度"""
    model.eval()
    with torch.no_grad():
        feat1 = model(series1)
        feat2 = model(series2)
    
    similarity = torch.nn.functional.cosine_similarity(feat1, feat2)
    return similarity.item()
```

```python
class SimilarityAnalyzer:
    """相似度分析器 - 支持多频率扩展"""
    
    def __init__(self, freq='daily', method='hybrid'):
        """
        freq: 数据频率 ('daily' 当前支持，其他计划中)
        method: 相似度计算方法
            - 'dtw': 动态时间规整
            - 'cosine': 余弦相似度
            - 'feature': 特征向量相似度
            - 'deep_learning': 深度学习相似度
            - 'hybrid': 混合方法（推荐）
        """
        self._validate_freq(freq)
        self.freq = freq
        self.method = method
        
        # 窗口配置
        self.window_map = {
            'daily': [20, 60, 120],       # 当前支持
            '5min': [48, 96, 192],        # 计划中
            '15min': [16, 32, 64],        # 计划中
            '60min': [20, 40, 80]         # 计划中
        }
        
        # 深度学习模型（延迟加载）
        self._dl_model = None
    
    def _validate_freq(self, freq):
        """验证频率"""
        supported = ['daily']
        planned = ['5min', '15min', '30min', '60min']
        
        if freq not in supported:
            if freq in planned:
                raise NotImplementedError(f"频率 {freq} 计划支持中")
            else:
                raise ValueError(f"不支持的频率: {freq}")
    
    def calculate_similarity(self, series1, series2):
        """计算相似度"""
        if self.method == 'dtw':
            return calculate_dtw_similarity(series1, series2)
        elif self.method == 'cosine':
            return cosine_similarity(series1, series2)
        elif self.method == 'deep_learning':
            return self._calculate_dl_similarity(series1, series2)
        elif self.method == 'hybrid':
            return self._calculate_hybrid_similarity(series1, series2)
    
    def _calculate_hybrid_similarity(self, series1, series2):
        """混合相似度：结合传统方法和深度学习"""
        # 传统方法
        dtw_sim = calculate_dtw_similarity(series1, series2)
        cosine_sim = cosine_similarity(series1, series2)
        
        # 深度学习（如果模型已加载）
        if self._dl_model is not None:
            dl_sim = calculate_dl_similarity(self._dl_model, series1, series2)
            return 0.3 * dtw_sim + 0.3 * cosine_sim + 0.4 * dl_sim
        else:
            return 0.5 * dtw_sim + 0.5 * cosine_sim
    
    def find_similar_stocks(self, target_stock, candidate_pool, window, top_k=10):
        """找出与目标股票走势相似的股票"""
        target_series = self.get_price_series(target_stock, window)
        
        similarities = []
        for stock in candidate_pool:
            candidate_series = self.get_price_series(stock, window)
            sim = self.calculate_similarity(target_series, candidate_series)
            similarities.append({
                'stock_code': stock,
                'similarity': sim,
                'window': window
            })
        
        # 排序返回Top K
        similarities.sort(key=lambda x: x['similarity'], reverse=True)
        return similarities[:top_k]
    
    def find_similar_periods(self, stock_code, history_data, current_window=20, threshold=0.8):
        """在历史数据中查找与当前走势相似的时间段"""
        current_pattern = self.get_recent_pattern(stock_code, current_window)
        
        similar_periods = []
        for i in range(len(history_data) - current_window):
            historical_pattern = history_data[i:i+current_window]
            similarity = self.calculate_similarity(current_pattern, historical_pattern)
            
            if similarity > threshold:
                similar_periods.append({
                    'start_date': history_data.index[i],
                    'end_date': history_data.index[i + current_window],
                    'similarity': similarity,
                    'subsequent_return_5d': self._calculate_subsequent_return(history_data, i + current_window, 5),
                    'subsequent_return_20d': self._calculate_subsequent_return(history_data, i + current_window, 20)
                })
        
        return sorted(similar_periods, key=lambda x: x['similarity'], reverse=True)
```

```python
    metrics['turnover'] = calculate_factor_turnover(factor_data)
    
    # 5. 最大回撤
    metrics['max_drawdown'] = calculate_max_drawdown(layer_returns)
    
    return metrics
```

```python
def risk_parity_weighting(factors):
    """风险平价加权，使各因子对组合风险贡献相等"""
    cov_matrix = np.cov(factors.T)
    
    def risk_parity_objective(weights):
        portfolio_var = np.dot(weights.T, np.dot(cov_matrix, weights))
        marginal_risk = np.dot(cov_matrix, weights) / np.sqrt(portfolio_var)
        risk_contrib = weights * marginal_risk
        target_risk = portfolio_var / len(weights)
        return np.sum((risk_contrib - target_risk) ** 2)
    
    from scipy.optimize import minimize
    result = minimize(risk_parity_objective, 
                     x0=np.ones(len(factors)) / len(factors),
                     method='SLSQP',
                     bounds=[(0, 1) for _ in factors],
                     constraints={'type': 'eq', 'fun': lambda w: np.sum(w) - 1})
    
    return result.x
```

```python
class OverfittingDetector:
    """过拟合检测器"""
    
    def __init__(self, 
                 sharpe_decay_threshold=0.5,
                 pbo_threshold=0.5):
        """
        sharpe_decay_threshold: 夏普比率衰减阈值
        pbo_threshold: 过拟合概率阈值
        """
        self.sharpe_decay_threshold = sharpe_decay_threshold
        self.pbo_threshold = pbo_threshold
    
    def detect_sharpe_decay(self, in_sample_metrics, out_of_sample_metrics):
        """检测夏普比率衰减"""
        is_sharpe = in_sample_metrics.get('sharpe_ratio', 0)
        oos_sharpe = out_of_sample_metrics.get('sharpe_ratio', 0)
        
        if is_sharpe <= 0:
            return {'decay': 1.0, 'is_overfitted': True}
        
        decay = (is_sharpe - oos_sharpe) / is_sharpe
        
        return {
            'in_sample_sharpe': is_sharpe,
            'out_of_sample_sharpe': oos_sharpe,
            'decay': decay,
            'is_overfitted': decay > self.sharpe_decay_threshold
        }
    
    def calculate_pbo(self, metrics_list, n_splits=10):
        """
        计算过拟合概率（Probability of Backtest Overfitting）
        使用组合对称交叉验证（CSCV）方法
        """
        from itertools import combinations
        
        n_strategies = len(metrics_list)
        
        # 将数据分成n_splits份
        # 对每种分割方式计算样本内外表现
        # 统计样本内最优策略在样本外表现不佳的概率
        
        # 简化实现
        pbo_scores = []
        
        for i in range(n_splits):
            # 随机分割
            train_idx = np.random.choice(len(metrics_list), size=len(metrics_list)//2, replace=False)
            test_idx = np.array([j for j in range(len(metrics_list)) if j not in train_idx])
            
            # 找样本内最优
            train_sharpes = [metrics_list[j]['sharpe_ratio'] for j in train_idx]
            best_train_idx = train_idx[np.argmax(train_sharpes)]
            
            # 检查样本外排名
            test_sharpes = [metrics_list[j]['sharpe_ratio'] for j in test_idx]
            best_test_idx = test_idx[np.argmax(test_sharpes)]
            
            # 如果样本内最优不是样本外最优，计入
            if best_train_idx != best_test_idx:
                pbo_scores.append(1)
            else:
                pbo_scores.append(0)
        
        pbo = np.mean(pbo_scores)
        
        return {
            'pbo': pbo,
            'is_overfitted': pbo > self.pbo_threshold
        }
    
    def parameter_sensitivity_analysis(self, base_params, param_ranges, evaluate_func):
        """
        参数敏感性分析
        检查参数微小变化对结果的影响
        """
        base_result = evaluate_func(**base_params)
        
        sensitivity_results = {}
        
        for param_name, param_range in param_ranges.items():
            results = []
            for param_value in param_range:
                params = base_params.copy()
                params[param_name] = param_value
                result = evaluate_func(**params)
                results.append({
                    'param_value': param_value,
                    'sharpe_ratio': result['sharpe_ratio']
                })
            
            # 计算敏感度（结果方差）
            sharpes = [r['sharpe_ratio'] for r in results]
            sensitivity = np.std(sharpes) / np.mean(sharpes) if np.mean(sharpes) != 0 else float('inf')
            
            sensitivity_results[param_name] = {
                'sensitivity': sensitivity,
                'results': results,
                'is_sensitive': sensitivity > 0.3  # 敏感度阈值
            }
        
        return sensitivity_results
    
    def comprehensive_check(self, 
                           in_sample_metrics, 
                           out_of_sample_metrics,
                           metrics_list=None,
                           base_params=None,
                           param_ranges=None,
                           evaluate_func=None):
        """综合过拟合检测"""
        
        results = {
            'sharpe_decay': self.detect_sharpe_decay(in_sample_metrics, out_of_sample_metrics),
            'overall_overfitted': False,
            'warnings': []
        }
        
        # PBO检测
        if metrics_list:
            results['pbo'] = self.calculate_pbo(metrics_list)
            if results['pbo']['is_overfitted']:
                results['warnings'].append('过拟合概率过高')
        
        # 参数敏感性
        if base_params and param_ranges and evaluate_func:
            results['sensitivity'] = self.parameter_sensitivity_analysis(
                base_params, param_ranges, evaluate_func
            )
            
            sensitive_params = [
                p for p, r in results['sensitivity'].items() 
                if r['is_sensitive']
            ]
            if sensitive_params:
                results['warnings'].append(f'参数敏感度过高: {sensitive_params}')
        
        # 综合判断
        results['overall_overfitted'] = (
            results['sharpe_decay']['is_overfitted'] or
            (results.get('pbo', {}).get('is_overfitted', False))
        )
        
        return results
```

```python
class AlertSystem:
    """预警系统"""
    
    def __init__(self, config=None):
        self.config = config or {}
        self.alert_history = []
        
        # 预警阈值配置
        self.thresholds = {
            'market_stage_change': True,           # 市场阶段变化
            'divergence_threshold': 0.3,           # 背离度阈值
            'sentiment_extreme': 0.8,              # 情绪极端值
            'factor_ic_decay': 0.5,                # IC衰减阈值
            'similarity_pattern': 0.85,            # 相似形态阈值
            'drawdown_warning': 0.1,               # 回撤预警阈值
        }
    
    def check_market_stage_alert(self, current_stage, previous_stage):
        """检查市场阶段变化预警"""
        alerts = []
        
        # 趋势变化
        if current_stage.get('trend') != previous_stage.get('trend'):
            alerts.append({
                'type': 'MARKET_STAGE_CHANGE',
                'level': 'HIGH',
                'message': f"市场趋势变化: {previous_stage.get('trend')} → {current_stage.get('trend')}",
                'timestamp': datetime.now(),
                'details': {
                    'previous': previous_stage,
                    'current': current_stage
                }
            })
        
        # 背离度预警
        divergence = abs(current_stage.get('divergence', 0))
        if divergence > self.thresholds['divergence_threshold']:
            alerts.append({
                'type': 'DIVERGENCE_ALERT',
                'level': 'MEDIUM',
                'message': f"指数个股背离度异常: {divergence:.2%}",
                'timestamp': datetime.now()
            })
        
        # 情绪极端
        sentiment = current_stage.get('sentiment', 0.5)
        if sentiment > self.thresholds['sentiment_extreme']:
            alerts.append({
                'type': 'SENTIMENT_EXTREME',
                'level': 'MEDIUM',
                'message': f"市场情绪过热: {sentiment:.2f}",
                'timestamp': datetime.now()
            })
        elif sentiment < (1 - self.thresholds['sentiment_extreme']):
            alerts.append({
                'type': 'SENTIMENT_EXTREME',
                'level': 'MEDIUM',
                'message': f"市场情绪过冷: {sentiment:.2f}",
                'timestamp': datetime.now()
            })
        
        return alerts
    
    def check_factor_alert(self, factor_metrics):
        """检查因子失效预警"""
        alerts = []
        
        for factor_name, metrics in factor_metrics.items():
            # IC衰减检测
            if 'ic_decay' in metrics:
                if metrics['ic_decay'] > self.thresholds['factor_ic_decay']:
                    alerts.append({
                        'type': 'FACTOR_IC_DECAY',
                        'level': 'HIGH',
                        'message': f"因子 {factor_name} IC衰减严重: {metrics['ic_decay']:.2%}",
                        'timestamp': datetime.now(),
                        'factor': factor_name
                    })
            
            # IC转负
            if metrics.get('ic_mean', 0) < 0:
                alerts.append({
                    'type': 'FACTOR_IC_NEGATIVE',
                    'level': 'MEDIUM',
                    'message': f"因子 {factor_name} IC转负: {metrics['ic_mean']:.4f}",
                    'timestamp': datetime.now(),
                    'factor': factor_name
                })
        
        return alerts
    
    def check_similarity_alert(self, similar_patterns):
        """检查相似形态预警"""
        alerts = []
        
        for pattern in similar_patterns:
            if pattern['similarity'] > self.thresholds['similarity_pattern']:
                # 查找历史后续表现
                avg_return = pattern.get('avg_subsequent_return', 0)
                
                alerts.append({
                    'type': 'SIMILAR_PATTERN',
                    'level': 'HIGH' if avg_return < 0 else 'MEDIUM',
                    'message': f"发现高相似度历史形态: {pattern['similarity']:.2%}, 历史后续收益: {avg_return:.2%}",
                    'timestamp': datetime.now(),
                    'details': pattern
                })
        
        return alerts
    
    def check_drawdown_alert(self, current_drawdown):
        """检查回撤预警"""
        alerts = []
        
        if abs(current_drawdown) > self.thresholds['drawdown_warning']:
            alerts.append({
                'type': 'DRAWDOWN_WARNING',
                'level': 'HIGH' if abs(current_drawdown) > 0.15 else 'MEDIUM',
                'message': f"组合回撤达到: {current_drawdown:.2%}",
                'timestamp': datetime.now()
            })
        
        return alerts
    
    def process_alerts(self, alerts):
        """处理预警（记录、通知）"""
        for alert in alerts:
            self.alert_history.append(alert)
            
            # 根据级别处理
            if alert['level'] == 'HIGH':
                self._send_high_priority_alert(alert)
            elif alert['level'] == 'MEDIUM':
                self._send_medium_priority_alert(alert)
    
    def _send_high_priority_alert(self, alert):
        """发送高优先级预警"""
        # 可以扩展为邮件、短信、微信等通知方式
        print(f"[HIGH ALERT] {alert['message']}")
    
    def _send_medium_priority_alert(self, alert):
        """发送中优先级预警"""
        print(f"[MEDIUM ALERT] {alert['message']}")
    
    def get_alert_summary(self, days=7):
        """获取预警汇总"""
        cutoff_time = datetime.now() - timedelta(days=days)
        recent_alerts = [
            a for a in self.alert_history 
            if a['timestamp'] > cutoff_time
        ]
        
        summary = {
            'total': len(recent_alerts),
            'by_type': {},
            'by_level': {'HIGH': 0, 'MEDIUM': 0, 'LOW': 0}
        }
        
        for alert in recent_alerts:
            alert_type = alert['type']
            summary['by_type'][alert_type] = summary['by_type'].get(alert_type, 0) + 1
            summary['by_level'][alert['level']] += 1
        
        return summary
```

```python
        lines = ["发现以下高相似度历史形态：\n"]
        
        for i, pattern in enumerate(patterns[:5], 1):
            lines.append(
                f"{i}. 相似度: {pattern['similarity']:.2%}, "
                f"历史日期: {pattern['start_date']}, "
                f"后续5日收益: {pattern.get('subsequent_return_5d', 0):.2%}"
            )

                f"{metrics.get('ic_ir', 0):.2f} | {metrics.get('layer_spread', 0):.2%} |"

    def _format_strategy_performance(self, backtest_results):
```

```python
class QuantAnalysisSystem:
    """量化分析系统主类"""
    
    def __init__(self, config=None):
        self.config = config or {}
        
        # 初始化各模块
        self.data_adapter = DailyDataAdapter()
        self.market_detector = MarketStageDetector()
        self.similarity_analyzer = SimilarityAnalyzer(freq='daily', method='hybrid')
        self.factor_selector = FactorSelector()
        self.risk_manager = RiskManagementSystem()
        self.alert_system = AlertSystem()
        self.report_generator = ReportGenerator()
        self.todo_manager = SystemTodoManager()
    
    def run_daily_analysis(self, date=None):
        """运行每日分析"""
        date = date or datetime.now().date()
        
        # 1. 获取数据
        index_data = self.data_adapter.get_index_data(date)
        stock_data = self.data_adapter.get_price_data(date)
        sector_data = self.data_adapter.get_sector_data(date)
        
        # 2. 市场阶段识别
        market_stage = self.market_detector.identify(index_data, stock_data, sector_data)
        
        # 3. 相似度分析
        similar_patterns = self.similarity_analyzer.find_similar_periods(
            stock_pool=self.config.get('watch_list', []),
            threshold=0.8
        )
        
        # 4. 因子分析
        factor_metrics = self.factor_selector.evaluate_factors(stock_data)
        optimal_factors = self.factor_selector.optimize(stock_data, market_stage)
        
        # 5. 风控检查
        risk_check = self.risk_manager.run_pre_trade_check(
            positions=self._get_current_positions(),
            industry_mapping=self.data_adapter.get_industry_mapping(),
            market_cap_data=self.data_adapter.get_market_cap_data()
        )
        
        # 6. 预警检测
        alerts = self._check_alerts(market_stage, factor_metrics, similar_patterns)
        self.alert_system.process_alerts(alerts)
        
        # 7. 生成报告
        report = self.report_generator.generate_report('daily', {
            'market_stage': market_stage,
            'sector_data': sector_data,
            'similar_patterns': similar_patterns,
            'factor_metrics': factor_metrics,
            'alerts': alerts
        })
        
        return {
            'market_stage': market_stage,
            'similar_patterns': similar_patterns,
            'factor_metrics': factor_metrics,
            'optimal_factors': optimal_factors,
            'risk_check': risk_check,
            'alerts': alerts,
            'report': report
        }
    
    def _check_alerts(self, market_stage, factor_metrics, similar_patterns):
        """检查各类预警"""
        alerts = []
        
        # 市场阶段预警
        alerts.extend(self.alert_system.check_market_stage_alert(
            market_stage, 
            self._get_previous_market_stage()
        ))
        
        # 因子失效预警
        alerts.extend(self.alert_system.check_factor_alert(factor_metrics))
        
        # 相似形态预警
        alerts.extend(self.alert_system.check_similarity_alert(similar_patterns))
        
        return alerts
```
